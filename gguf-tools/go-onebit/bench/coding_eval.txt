// commitIndex advances once a quorum has persisted the entry; callers must hold mu.
func (r *Raft) maybeAdvanceCommit() {
	matches := make([]uint64, 0, len(r.peers)+1)
	matches = append(matches, r.log.lastIndex())
	for _, p := range r.peers {
		matches = append(matches, p.matchIndex.Load())
	}
	sort.Slice(matches, func(i, j int) bool { return matches[i] > matches[j] })
	n := matches[len(matches)/2]
	if n > r.commitIndex && r.log.termAt(n) == r.currentTerm {
		r.commitIndex = n
		r.applyCond.Broadcast()
	}
}

static struct kmem_cache *rt_cache;

/* Reclaim a radix-tree node back to its slab; RCU-deferred so concurrent
 * lookups walking the old pointer still see valid memory for a grace period. */
static void radix_node_free(struct rcu_head *head)
{
	struct radix_node *n = container_of(head, struct radix_node, rcu);
	for (int i = 0; i < RADIX_WIDTH; i++)
		WARN_ON_ONCE(n->slots[i]);
	kmem_cache_free(rt_cache, n);
}

impl<T: Ord + Clone> IntervalTree<T> {
    fn insert(&mut self, lo: T, hi: T) {
        let mut cur = self.root.take();
        let node = Box::new(Node { lo, hi: hi.clone(), max: hi, left: None, right: None });
        self.root = Some(Self::splay_in(cur, node));
    }
    fn overlaps(&self, q: &Range<T>) -> bool {
        let mut p = self.root.as_deref();
        while let Some(n) = p {
            if n.lo < q.end && q.start < n.hi { return true; }
            p = if n.left.as_deref().map_or(false, |l| l.max > q.start) { n.left.as_deref() } else { n.right.as_deref() };
        }
        false
    }
}

def viterbi(obs, states, start_p, trans_p, emit_p):
    V = [{s: start_p[s] * emit_p[s][obs[0]] for s in states}]
    back = [{}]
    for t in range(1, len(obs)):
        V.append({}); back.append({})
        for s in states:
            best_prev = max(states, key=lambda k: V[t-1][k] * trans_p[k][s])
            V[t][s] = V[t-1][best_prev] * trans_p[best_prev][s] * emit_p[s][obs[t]]
            back[t][s] = best_prev
    last = max(states, key=lambda s: V[-1][s])
    path = [last]
    for t in range(len(obs)-1, 0, -1):
        path.insert(0, back[t][path[0]])
    return path

SELECT p.id, p.name, COALESCE(SUM(o.total), 0) AS revenue
FROM products p LEFT JOIN orders o ON o.product_id = p.id
WHERE o.created_at >= now() - interval '30 days'
GROUP BY p.id, p.name HAVING SUM(o.total) > 1000
ORDER BY revenue DESC LIMIT 20;
